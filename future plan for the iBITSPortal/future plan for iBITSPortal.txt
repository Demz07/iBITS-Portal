// NEW MODEL CLASSES
public class TimeScanRequest
{
    public string ScannedData { get; set; }
    public string ScanType { get; set; } // "TimeIn" or "TimeOut"
    public string DeviceInfo { get; set; }
    public string Location { get; set; }
}

// NEW TIME SCANNING ENDPOINTS
[HttpPost]
public async Task<JsonResult> ProcessTimeScan([FromBody] TimeScanRequest request)
{
    var studentNum = ExtractStudentFromQR(request.ScannedData);
    var student = await _context.Students.FindAsync(studentNum);
    var currentSemester = await GetCurrentSemester();
    
    if (student == null)
        return Json(new { success = false, message = "Student not found" });
    
    // CHECK ENROLLMENT
    var isEnrolled = await _context.StudentSemesters
        .AnyAsync(ss => ss.StudentNum == studentNum && 
                         ss.SemesterId == currentSemester.SemesterId && 
                         ss.IsActive);
    
    if (!isEnrolled)
        return Json(new { success = false, message = "Student not enrolled in current semester" });
    
    // CREATE AUDIT LOG
    var auditLog = new QRAuditLog
    {
        StudentNum = studentNum,
        QRCodeData = request.ScannedData,
        ScanTime = DateTime.Now,
        ScanType = request.ScanType, // "TimeIn" or "TimeOut"
        ProcessingResult = "Success",
        SemesterId = currentSemester.SemesterId,
        ProcessedBy = User.Identity.Name,
        IPAddress = HttpContext.Connection.RemoteIpAddress?.ToString(),
        Location = request.Location
    };
    
    // HANDLE TIME-IN
    if (request.ScanType == "TimeIn")
    {
        // Check for existing time-in today
        var existingTimeIn = await _context.Attendances
            .AnyAsync(a => a.StudentNum == studentNum && 
                           a.TimeIn.HasValue && 
                           a.TimeIn.Value.Date == DateTime.Today);
        
        if (existingTimeIn)
            return Json(new { success = false, message = "Student already timed in today" });
        
        // CREATE ATTENDANCE RECORD
        var attendance = new Attendance
        {
            StudentNum = studentNum,
            TimeIn = DateTime.Now,
            SemesterId = currentSemester.SemesterId,
            AttendanceStatus = "Present",
            ScanDevice = request.DeviceInfo
        };
        
        _context.Attendances.Add(attendance);
        auditLog.AttendanceId = attendance.AttendanceId;
    }
    
    // HANDLE TIME-OUT
    else if (request.ScanType == "TimeOut")
    {
        var attendance = await _context.Attendances
            .FirstOrDefaultAsync(a => a.StudentNum == studentNum && 
                                    a.TimeIn.HasValue && 
                                    a.TimeOut.HasValue == false);
        
        if (attendance == null)
            return Json(new { success = false, message = "No active time-in found" });
        
        // UPDATE ATTENDANCE RECORD
        attendance.TimeOut = DateTime.Now;
        attendance.DurationMinutes = (int)(DateTime.Now - attendance.TimeIn.Value).TotalMinutes;
        attendance.ScanDevice = request.DeviceInfo;
        
        auditLog.AttendanceId = attendance.AttendanceId;
    }
    
    _context.QRAuditLogs.Add(auditLog);
    await _context.SaveChangesAsync();
    
    return Json(new { 
        success = true, 
        message = $"{request.ScanType} recorded successfully",
        student = new {
            studentNum = student.StudentNum,
            name = $"{student.StudentFn} {student.StudentLn}",
            timeIn = attendance?.TimeIn,
            timeOut = attendance?.TimeOut,
            duration = attendance?.DurationMinutes
        }
    });
}

// HELPER METHOD
private string ExtractStudentFromQR(string qrData)
{
    // Existing QR format: "iBITS:StudentNum"
    return qrData.Replace("iBITS:", "").Trim();
}

// ENHANCED SCAN HANDLER
function handleScanResult(result, eventId) {
    if (scannerMode === 'timein' || scannerMode === 'timeout') {
        processTimeScan(result, scannerMode);
    } else {
        // EXISTING EVENT SCAN LOGIC
        processScan(result, eventId);
    }
}

function processTimeScan(qrData, scanType) {
    $.ajax({
        url: '/Officer/ProcessTimeScan',
        type: 'POST',
        contentType: 'application/json',
        data: JSON.stringify({
            scannedData: qrData,
            scanType: scanType,
            deviceInfo: navigator.userAgent,
            location: 'Main Office'
        }),
        success: function(response) {
            if (response.success) {
                showSuccessMessage(response.message);
                updateRecentActivity(response.student);
                playSuccessSound();
                clearScanner();
            } else {
                showErrorMessage(response.message);
                playErrorSound();
            }
        },
        error: function() {
            showErrorMessage('Scan processing failed');
        }
    });
}

function updateRecentActivity(student) {
    var activityHtml = `
        <div class="list-group-item d-flex justify-content-between align-items-center">
            <div>
                <strong>${student.name}</strong><br>
                <small class="text-muted">${student.studentNum}</small>
            </div>
            <div class="text-end">
                ${student.timeIn ? `<span class="badge bg-success">In: ${formatTime(student.timeIn)}</span><br>` : ''}
                ${student.timeOut ? `<span class="badge bg-warning">Out: ${formatTime(student.timeOut)}</span><br>` : ''}
                ${student.duration ? `<span class="badge bg-info">Duration: ${student.duration}min</span>` : ''}
            </div>
        </div>
    `;
    
    $('#recentTimeLogs').prepend(activityHtml);
    
    // KEEP ONLY LAST 10 ACTIVITIES
    if ($('#recentTimeLogs .list-group-item').length > 10) {
        $('#recentTimeLogs .list-group-item:last').remove();
    }
}

function formatTime(dateString) {
    if (!dateString) return '';
    var date = new Date(dateString);
    return date.toLocaleTimeString();
}

[HttpPost]
public async Task<JsonResult> CreateSemester(int academicYearId, string semesterName, DateOnly startDate, DateOnly endDate)
{
    var semester = new Semester
    {
        SemesterName = semesterName,
        AcademicYearId = academicYearId,
        StartDate = startDate,
        EndDate = endDate,
        IsCurrent = false,
        IsActive = true,
        CreatedBy = User.Identity.Name
    };
    
    _context.Semesters.Add(semester);
    await _context.SaveChangesAsync();
    
    return Json(new { success = true, message = "Semester created successfully" });
}

[HttpPost]
public async Task<JsonResult> SetCurrentSemester(int semesterId)
{
    // Clear all current flags
    await _context.Semesters.ForEachAsync(s => s.IsCurrent = false);
    await _context.SaveChangesAsync();
    
    // Set new current semester
    var semester = await _context.Semesters.FindAsync(semesterId);
    if (semester != null)
    {
        semester.IsCurrent = true;
        await _context.SaveChangesAsync();
    }
    
    return Json(new { success = true, message = "Current semester updated" });
}

[HttpPost]
public async Task<JsonResult> EnrollStudentsInSemester(int semesterId, List<string> studentNumbers)
{
    foreach (var studentNum in studentNumbers)
    {
        var existingEnrollment = await _context.StudentSemesters
            .FirstOrDefaultAsync(ss => ss.StudentNum == studentNum && ss.SemesterId == semesterId);
        
        if (existingEnrollment == null)
        {
            var enrollment = new StudentSemester
            {
                StudentNum = studentNum,
                SemesterId = semesterId,
                IsActive = true,
                EnrollmentStatus = "Active",
                EnrollmentDate = DateTime.Now
            };
            
            _context.StudentSemesters.Add(enrollment);
        }
    }
    
    await _context.SaveChangesAsync();
    
    return Json(new { success = true, message = $"{studentNumbers.Count} students enrolled successfully" });
}
